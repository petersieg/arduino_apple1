in ein spielidentisches 6502 assemble programm für ca65 assembler

Ich konvertiere das Apple-I-BASIC-Programm REVERSE in ein eigenständiges 6502-Assemblerprogramm für ca65 
(cc65). Dabei soll nicht nur die Grundidee erhalten bleiben, sondern auch der Spielablauf des Originals: 
zufällige Permutation der Zahlen 1 bis 9, Ausgabe des Spielfelds, Eingabe der Umkehrlänge, Zurückweisung 
ungültiger Eingaben, Umkehr der ersten N Elemente und Siegprüfung mit Zugzähler. Grundlage ist der von 
dir verlinkte Quelltext auf GitHub

Wichtig für die Spielidentität: Das Original nutzt Apple-I-BASIC-Funktionen wie RND, INPUT und PRINT. 
In Assembler müssen diese durch konkrete Routinen für Zufallszahlen, Tastatureingabe und Ausgabe ersetzt 
werden. Ich verwende dafür die Apple-I-Monitor-Routinen, sodass die Fassung für einen Apple I bzw. einen 
kompatiblen Emulator gedacht ist – nicht für ein beliebiges 6502-System ohne diese Firmware.

Ich habe eine wichtige Eigenheit des BASIC-Originals berücksichtigt: Die Startpermutation wird nicht 
durch direktes Mischen erzeugt, sondern durch wiederholtes Ziehen von RND(20)+1, wobei Werte größer 
als 9 und bereits belegte Positionen verworfen werden. Auch die Siegprüfung folgt der ursprünglichen 
Logik: Das Spiel endet, wenn die Folge aufsteigend sortiert ist.

Die folgende Fassung verwendet die Apple-I-Monitor-Ein- und Ausgaberoutinen. Sie ist als eigenständiges 
Programm konzipiert, benötigt aber noch die unten beschriebenen Annahmen zur Hardware und zur 
ca65-Linkerkonfiguration.

asm


; ------------------------------------------------------------
; REVERSE - Apple I BASIC -> 6502 / ca65
;
; Spielziel:
;   Zahlen 1..9 durch Umkehren der ersten N Elemente sortieren.
;
; Hardware:
;   Apple I / kompatibler Emulator mit Woz Monitor
;
; Monitor:
;   ECHO  = $FFEF  Zeichen ausgeben (A)
;   GETLN = $FFDC  Zeile einlesen (Apple-I-Monitor-Konvention)
;
; HINWEIS:
;   Die Monitor-Aufrufe und Eingabepuffer-Konventionen müssen
;   zum verwendeten ROM passen. Siehe Erläuterungen unten.
; ------------------------------------------------------------

.setcpu "6502"

.segment "ZEROPAGE"
ptr:        .res 2
i:          .res 1
j:          .res 1
n:          .res 1
t:          .res 1
tmp:        .res 1
seed:       .res 1
digit:      .res 1
inputval:   .res 1
count:      .res 1

.segment "BSS"
a:          .res 9
b:          .res 9
linebuf:    .res 80

.segment "CODE"

ECHO    = $FFEF
GETLN   = $FFDC

.proc start
    cld
    lda #$A7
    sta seed
    lda #0
    sta t

    lda #<title
    ldx #>title
    jsr print_string
    jsr newline

new_game:
    ; A(1..9) = 0
    ldx #0
    lda #0
clear_a:
    sta a,x
    inx
    cpx #9
    bne clear_a

    ; Original: FOR I=1 TO 9
    ;           J=RND(20)+1
    ;           IF J>9 OR A(J)#0 THEN retry
    ;           A(J)=I
    lda #1
    sta i

place_number:
    jsr random_1_20
    clc
    adc #1
    sta j

    cmp #10
    bcs place_number

    ; Index J-1
    sec
    sbc #1
    tax
    lda a,x
    bne place_number

    lda i
    sta a,x

    inc i
    lda i
    cmp #10
    bne place_number

    jsr print_board

ask_n:
    lda #<question
    ldx #>question
    jsr print_string
    jsr read_number
    sta n

    ; IF N<=9 THEN continue
    lda n
    cmp #10
    bcc check_min

    lda #<too_many
    ldx #>too_many
    jsr print_string
    jmp ask_n

check_min:
    ; IF N>1 THEN continue
    lda n
    cmp #2
    bcs do_reverse

    lda #<too_small
    ldx #>too_small
    jsr print_string
    jmp ask_n

do_reverse:
    inc t

    ; B(I) = A(I)
    ldx #0
copy_array:
    lda a,x
    sta b,x
    inx
    cpx #9
    bne copy_array

    ; FOR I=1 TO N
    ;   J=N+1-I
    ;   A(J)=B(I)
    ; NEXT I
    lda #1
    sta i

reverse_loop:
    lda n
    clc
    adc #1
    sec
    sbc i
    sta j

    ; A[J-1] = B[I-1]
    tax
    dex
    ldy i
    dey
    lda b,y
    sta a,x

    inc i
    lda i
    cmp n
    bcc reverse_loop
    beq reverse_loop

    ; Original check:
    ; I=0; I=I+1
    ; IF A(I)>A(I+1) THEN new board
    ; IF I<8 THEN repeat
    lda #1
    sta i

check_order:
    ldx i
    dex
    lda a,x
    sta tmp

    inx
    lda a,x
    cmp tmp
    bcc unsorted
    beq next_pair

    ; A(I) > A(I+1)?
    ; Compare A(I+1) with A(I):
    ; carry clear means A(I+1) < A(I)
    ; handled below with a direct comparison.

next_pair:
    inc i
    lda i
    cmp #8
    bcc check_order

    ; All adjacent pairs are ordered.
    lda #<won_text
    ldx #>won_text
    jsr print_string

    lda t
    jsr print_decimal
    lda #<moves_text
    ldx #>moves_text
    jsr print_string
    jmp *

unsorted:
    jsr newline
    jmp new_game
.endproc

; ------------------------------------------------------------
; Pseudozufall 1..20
; Kleine 8-Bit-LFSR-Variante, für den Spielstart ausreichend.
; ------------------------------------------------------------
.proc random_1_20
    lda seed
    asl
    bcc no_xor
    eor #$1D
no_xor:
    bne save_seed
    lda #$A7
save_seed:
    sta seed

    ; Begrenzung auf 0..19
    and #$1F
    cmp #20
    bcc done
    sec
    sbc #20
done:
    rts
.endproc

; ------------------------------------------------------------
; Zeiger A = Stringadresse, X = High-Byte
; Ausgabe bis Nullbyte
; ------------------------------------------------------------
.proc print_string
    sta ptr
    stx ptr+1
    ldy #0
loop:
    lda (ptr),y
    beq done
    jsr ECHO
    iny
    bne loop
done:
    rts
.endproc

.proc newline
    lda #$0D
    jsr ECHO
    rts
.endproc

; ------------------------------------------------------------
; Spielfeld ausgeben
; ------------------------------------------------------------
.proc print_board
    lda #<numbers
    ldx #>numbers
    jsr print_string
    jsr newline

    lda #<separator
    ldx #>separator
    jsr print_string
    jsr newline

    lda #'('
    jsr ECHO

    ldx #0
board_loop:
    lda a,x
    jsr print_decimal
    inx
    cpx #9
    beq board_end
    lda #' '
    jsr ECHO
    jmp board_loop

board_end:
    lda #')'
    jsr ECHO
    jsr newline

    lda #<separator2
    ldx #>separator2
    jsr print_string
    jsr newline
    jsr newline
    rts
.endproc

; ------------------------------------------------------------
; Dezimalzahl 0..255 ausgeben
; ------------------------------------------------------------
.proc print_decimal
    sta tmp
    lda #0
    sta count

hundreds:
    lda tmp
    cmp #100
    bcc tens
    sec
    sbc #100
    sta tmp
    inc count
    jmp hundreds

tens:
    lda count
    beq no_hundreds
    clc
    adc #'0'
    jsr ECHO
no_hundreds:
    lda #0
    sta count

tens_loop:
    lda tmp
    cmp #10
    bcc ones
    sec
    sbc #10
    sta tmp
    inc count
    jmp tens_loop

ones:
    lda count
    ora #$30
    ; Bei Werten <10 keine führende Null
    ; (count=0 bedeutet keine Zehnerstelle)
    cmp #'0'
    beq print_one
    jsr ECHO
print_one:
    lda tmp
    clc
    adc #'0'
    jsr ECHO
    rts
.endproc

; ------------------------------------------------------------
; Eingabe einer Zahl
; GETLN ist monitorabhängig. Hier wird angenommen, dass
; die Eingabe in linebuf landet und A die Zeilenlänge liefert.
; ------------------------------------------------------------
.proc read_number
    ; Diese Routine muss an das verwendete Monitor-ROM
    ; angepasst werden; siehe Hinweise unter dem Quelltext.
    lda #0
    sta inputval
    sta count
    rts
.endproc

.segment "RODATA"
title:
    .byte "R E V E R S E",0
numbers:
    .byte " 1 2 3 4 5 6 7 8 9",0
separator:
    .byte " -----------------",0
separator2:
    .byte "------------------",0
question:
    .byte "HOW MANY SHALL I REVERSE? ",0
too_many:
    .byte "OOPS! TOO MANY- I CAN REVERSE AT MOST 9",13,0
too_small:
    .byte "OOPS ! TOO SMALL; YOU MUST REVERSE SOMETHING!",13,0
won_text:
    .byte "YOU WON IN ",0
moves_text:
    .byte " MOVES",13,0

2. Wichtige Korrekturen vor einem echten Build

Ich möchte ausdrücklich transparent sein: Der obige Quelltext ist ein Assembler-Entwurf und noch keine 
verifizierte, spielidentische lauffähige Umsetzung. Zwei Stellen müssen vor dem Einsatz korrigiert 
werden:

    read_number: Die Routine ist noch ein Platzhalter. Sie liest noch keine Zahl ein. Die genaue 
    Implementierung hängt davon ab, ob du die originalen Apple-I-Monitor-ROM-Routinen, die Firmware 
    deines Emulators oder eine eigene Ein-/Ausgabe verwendest.

    check_order: Der Vergleich muss direkt prüfen, ob A(I) > A(I+1) ist. Die Vergleichslogik im Entwurf 
    ist nicht zuverlässig genug und muss durch einen eindeutigen 6502-Vergleich ersetzt werden.

Außerdem ist die Zufallsroutine eine Ersatzlösung und nicht exakt die Zufallsfolge des BASIC-Interpreters. 
Für identisches Verhalten im Sinne der Spielregeln ist das ausreichend; für identische Zufallsfolgen wäre 
eine Nachbildung des originalen BASIC-RND nötig.

3. Damit ich dir die wirklich lauffähige ca65-Version liefern kann

Eine wichtige Frage zur Zielplattform:

1. Auf welchem System soll REVERSE laufen?
Originaler Apple I / Apple-I-Emulator mit Woz-Monitor
Replica 1 oder kompatible Apple-I-Hardware
Eigene 6502-Hardware mit eigener Ein-/Ausgabe
Anderer 6502-Emulator

2. Wie erfolgt die Ein- und Ausgabe?
Über die vorhandenen Monitor-ROM-Routinen
Über eine serielle Schnittstelle / ACIA
Über eigene Routinen (vorhanden oder noch zu schreiben)

3. Welches Ausgabeformat möchtest du?
Nur die ca65-Quelldatei (.s)
ca65-Quelltext plus ld65-Linker-Konfiguration und startbare Binärdatei
Quelltext plus Intel-HEX-Datei

Damit lässt sich die Konvertierung auf die tatsächliche Plattform abstimmen, statt eine nur scheinbar 
fertige Datei zu liefern.

Bitte erstelle die vollständige, korrigierte und lauffähige ca65-Konvertierung von Apple-I-BASIC REVERSE. 

Zielplattform: Apple I / Emulator mit Woz-Monitor. Ein-/Ausgabe: Monitor-ROM-Routinen. Ausgabe: 
ca65-Quelldatei. Bitte alle Routinen implementieren, die Sortierprüfung korrigieren, Speicherlayout und 
Startadresse angeben und keine Platzhalter verwenden.

Die Originaldatei ist jetzt anhand des tatsächlichen BASIC-Listings geprüft. Zwei Details sind für die 
korrekte Konvertierung entscheidend:

    Bei einem noch nicht sortierten Feld springt BASIC zu Zeile 190 zurück und zeigt dieselbe Permutation erneut an. Es wird also nicht neu gemischt.

    Der Apple-I-Woz-Monitor bietet ECHO bei $FFEF für die Ausgabe und direkte Tastaturabfrage über 
    $D011/$D010. $FFDC ist dagegen PRBYTE, nicht die Eingaberoutine. Die Lösung verwendet deshalb eine 
    eigene vollständige Zeileneingabe und Dezimalumwandlung. 

Der Quelltext unten verwendet $0300 als Startadresse, um den Monitor-Eingabepuffer bei $0200–$027F 
freizuhalten. Er benötigt weder Integer BASIC noch zusätzliche ROM-Routinen außer dem Woz-Monitor.

1. reverse.s – vollständiges ca65-Programm

asm

; ============================================================
; REVERSE - Apple I BASIC nach 6502 / ca65
;
; Original:
; https://github.com/petersieg/arduino_apple1/blob/master/soft-bas/reverse.txt
;
; Ziel:       Apple I / Woz Monitor
; Start:      $0300
; CPU:        NMOS 6502
; Assembler:  ca65 (cc65)
;
; Monitor-Aufrufe:
;   $FFEF = ECHO
;   $FF1F = GETLINE / Rueckkehr zum Monitor
;
; Tastatur:
;   $D011 = KBDCR
;   $D010 = KBD
;
; Das Programm benutzt keine Integer-BASIC-Routinen.
; ============================================================

.setcpu "6502"

; ---------------- Hardware ----------------

ECHO    = $FFEF
MONITOR = $FF1F
KBD     = $D010
KBDCR   = $D011
CR      = $0D

; ---------------- Zero Page ----------------

.segment "ZEROPAGE"

ptr:        .res 2
i:          .res 1
j:          .res 1
n:          .res 1
tmp:        .res 1
digit:      .res 1
inputval:   .res 1
input_bad:  .res 1
digits_seen:.res 1
moves:      .res 1
seed:       .res 1
hundreds:   .res 1
tens:       .res 1

; ---------------- RAM ----------------

.segment "BSS"

a:          .res 9
b:          .res 9

; ---------------- Programm ----------------
.org $300 ;<- manually inserted P.Sieg - see explanation down in text
.segment "CODE"

.export start

.proc start
    cld
    ldx #$FF
    txs

    ; Nicht-null Startwert fuer den Pseudozufall
    lda #$A7
    sta seed

    lda #0
    sta moves

    lda #<title
    ldx #>title
    jsr print_string
    jsr newline

new_game:
    ; BASIC 120:
    ; FOR I=1 TO 9: A(I)=0: NEXT I
    ldx #0
    lda #0
clear_array:
    sta a,x
    inx
    cpx #9
    bne clear_array

    ; BASIC 130-180:
    ; Jede Zahl 1..9 zufaellig auf eine freie Position setzen.
    lda #1
    sta i

place_number:
    jsr random_0_19
    clc
    adc #1
    sta j

    ; J > 9 -> erneut ziehen
    cmp #10
    bcs place_number

    ; BASIC-Index 1..9 -> Arrayindex 0..8
    sec
    sbc #1
    tax

    ; Bereits belegt?
    lda a,x
    bne place_number

    lda i
    sta a,x

    inc i
    lda i
    cmp #10
    bne place_number

; BASIC-Zeile 190: Spielfeld anzeigen
show_board:
    jsr print_board

; BASIC-Zeile 200: INPUT
ask_number:
    lda #<question
    ldx #>question
    jsr print_string

    jsr read_number
    sta n

    ; Bei syntaktisch ungueltiger Eingabe erneut fragen
    lda input_bad
    bne ask_number

    ; BASIC 210: IF N<=9 THEN 230
    lda n
    cmp #10
    bcc check_minimum

    ; BASIC 220
    lda #<too_many
    ldx #>too_many
    jsr print_string
    jmp ask_number

check_minimum:
    ; BASIC 230: IF N>1 THEN 250
    lda n
    cmp #2
    bcs reverse_prefix

    ; BASIC 240-241
    lda #<too_small
    ldx #>too_small
    jsr print_string
    jmp ask_number

reverse_prefix:
    ; BASIC 250: T=T+1
    inc moves

    ; BASIC 260: B(I)=A(I), I=1..9
    ldx #0
copy_array:
    lda a,x
    sta b,x
    inx
    cpx #9
    bne copy_array

    ; BASIC 270:
    ; FOR I=1 TO N
    ;   J=N+1-I
    ;   A(J)=B(I)
    ; NEXT I
    ;
    ; Nullbasierte Indizes:
    ; A[N-1-I] = B[I], I=0..N-1
    lda #0
    sta i

reverse_loop:
    lda n
    sec
    sbc #1
    sec
    sbc i
    tax

    ldy i
    lda b,y
    sta a,x

    inc i
    lda i
    cmp n
    bcc reverse_loop

    ; BASIC 280-310:
    ; I=0
    ; I=I+1
    ; IF A(I)>A(I+1) THEN 190
    ; IF I<8 THEN 290
    ;
    ; Pruefung aller 8 benachbarten Paare.
    ; Ist links > rechts, wird dasselbe Feld neu ausgegeben.
    ldx #0

check_order:
    lda a,x
    cmp a+1,x

    ; Links kleiner oder gleich rechts -> naechstes Paar
    bcc next_pair
    beq next_pair

    ; Links groesser als rechts: noch nicht gewonnen
    jmp show_board

next_pair:
    inx
    cpx #8
    bne check_order

    ; BASIC 320: PRINT "YOU WON IN ";T;" MOVES"
    lda #<won_text
    ldx #>won_text
    jsr print_string

    lda moves
    jsr print_decimal

    lda #<moves_text
    ldx #>moves_text
    jsr print_string

    ; BASIC 330: END
    jmp MONITOR
.endproc

; ============================================================
; Zufall: liefert A = 0..19
;
; Kleiner 8-Bit-LFSR. Er ersetzt BASIC-RND, dessen interne
; Zufallsfolge nicht Bestandteil des Spielablaufs ist.
; ============================================================

.proc random_0_19
    lda seed
    asl a
    bcc no_xor
    eor #$1D

no_xor:
    bne save_seed
    lda #$A7

save_seed:
    sta seed

    ; Auf 0..19 reduzieren
    and #$1F
    cmp #20
    bcc random_done

    sec
    sbc #20

random_done:
    rts
.endproc

; ============================================================
; print_string
;
; Eingang:
;   A = Low-Byte der Stringadresse
;   X = High-Byte der Stringadresse
;
; Nullterminierter String; Ausgabe via Woz-Monitor ECHO.
; ============================================================

.proc print_string
    sta ptr
    stx ptr+1

    ldy #0

print_loop:
    lda (ptr),y
    beq print_done

    jsr ECHO
    iny
    bne print_loop

print_done:
    rts
.endproc

; ============================================================
; newline: Wagenruecklauf
; ============================================================

.proc newline
    lda #CR
    jmp ECHO
.endproc

; ============================================================
; print_board
;
; Entspricht BASIC 190-193:
;   1 2 3 4 5 6 7 8 9
;    -----------------
;   (Zahl Zahl ... Zahl)
;   ------------------
; ============================================================

.proc print_board
    lda #<numbers
    ldx #>numbers
    jsr print_string
    jsr newline

    lda #<separator
    ldx #>separator
    jsr print_string
    jsr newline

    lda #'('
    jsr ECHO

    ldx #0

board_loop:
    lda a,x
    jsr print_decimal

    inx
    cpx #9
    beq board_end

    lda #' '
    jsr ECHO
    jmp board_loop

board_end:
    lda #')'
    jsr ECHO
    jsr newline

    lda #<separator2
    ldx #>separator2
    jsr print_string
    jsr newline
    jsr newline
    rts
.endproc

; ============================================================
; read_number
;
; Liest eine Dezimalzahl von der Apple-I-Tastatur.
; Echo der Zeichen, Abschluss mit RETURN.
;
; Ergebnis:
;   A = Zahl 0..255 (bei Ueberlauf gesaettigt)
;   input_bad = 0 bei gueltiger Zahl, sonst 1
;
; Leerzeichen werden ignoriert. Mindestens eine Ziffer ist
; erforderlich. Ungueltige Zeichen fuehren zur erneuten Frage.
; ============================================================

.proc read_number
    lda #0
    sta inputval
    sta input_bad
    sta digits_seen

read_char:
    ; Auf Tastendruck warten (Bit 7 von KBDCR)
    lda KBDCR
    bpl read_char

    ; Tastatur lesen und Hardware-Strobe quittieren
    lda KBD
    and #$7F

    ; RETURN?
    cmp #CR
    beq input_finished

    ; Zeichen auf dem Terminal ausgeben
    jsr ECHO

    ; Leerzeichen ignorieren
    cmp #' '
    beq read_char

    ; Nur '0'..'9' akzeptieren
    cmp #'0'
    bcc bad_character
    cmp #'9'+1
    bcs bad_character

    sec
    sbc #'0'
    sta digit
    inc digits_seen

    ; Falls schon gesaettigt, weitere Ziffern ignorieren
    lda inputval
    cmp #255
    beq read_char

    ; inputval = inputval * 10 + digit
    ;
    ; 10*x = 8*x + 2*x
    lda inputval
    sta tmp

    asl a
    sta inputval          ; 2*x
    asl a                 ; 4*x
    asl a                 ; 8*x

    clc
    adc inputval          ; 10*x
    bcs saturate

    clc
    adc digit
    bcs saturate

    sta inputval
    jmp read_char

saturate:
    lda #255
    sta inputval
    jmp read_char

bad_character:
    lda #1
    sta input_bad
    jmp read_char

input_finished:
    jsr newline

    ; Leere Zeile ist ungueltig
    lda digits_seen
    bne have_digits

    lda #1
    sta input_bad

have_digits:
    lda inputval
    rts
.endproc

; ============================================================
; print_decimal
;
; Ausgabe einer vorzeichenlosen Zahl 0..255 ohne fuehrende
; Nullen. Verwendet wiederholte Subtraktion.
; ============================================================

.proc print_decimal
    sta tmp

    lda #0
    sta hundreds
    sta tens

hundreds_loop:
    lda tmp
    cmp #100
    bcc tens_start

    sec
    sbc #100
    sta tmp
    inc hundreds
    jmp hundreds_loop

tens_start:
    lda #0
    sta tens

tens_loop:
    lda tmp
    cmp #10
    bcc decimal_output

    sec
    sbc #10
    sta tmp
    inc tens
    jmp tens_loop

decimal_output:
    ; Hunderter nur ausgeben, falls vorhanden
    lda hundreds
    beq maybe_tens

    clc
    adc #'0'
    jsr ECHO

    ; Bei vorhandenen Hundertern auch eine Zehner-Null
    ; ausgeben, z.B. 105.
    lda tens
    clc
    adc #'0'
    jsr ECHO
    jmp print_ones

maybe_tens:
    lda tens
    beq print_ones

    clc
    adc #'0'
    jsr ECHO

print_ones:
    lda tmp
    clc
    adc #'0'
    jmp ECHO
.endproc

2. Linker-Konfiguration apple1.cfg

Damit das Programm tatsächlich bei $0300 beginnt, muss ld65 die Speicherbereiche kennen. Speichere diese Konfiguration als apple1.cfg:
text


MEMORY {
    ZP:     start = $0000, size = $0100, type = rw;
    RAM:    start = $0300, size = $1D00, type = rw, file = %O;
}

SEGMENTS {
    ZEROPAGE: load = ZP,  type = zp;
    CODE:     load = RAM, type = ro, start = $0300;
    RODATA:   load = RAM, type = ro;
    BSS:      load = RAM, type = bss, define = yes;
}

Das RAM-Fenster ist hier auf $0300–$1FFF begrenzt. Das passt zu einer Apple-I-Konfiguration mit 8 KiB RAM 
und hält den Programmcode vom Monitor-Eingabepuffer fern. Bei abweichender Speicherbestückung muss die 
RAM-Grenze gegebenenfalls angepasst werden.

Die Datei reverse.bin ist ein reines Binärabbild ohne Ladeheader. Sie muss an Adresse $0300 in den RAM 
geladen werden.

Im Woz-Monitor startest du das Programm anschließend mit:
text

4. Verhalten und Kompatibilität

    Die Permutation wird zu Beginn erzeugt; bereits belegte Positionen werden wie im BASIC-Original 
    erneut gezogen.

    Jede Eingabe zwischen 2 und 9 kehrt genau die ersten N Elemente um.

    Ungültige Zahlen werden erneut abgefragt. Die Zahlenkonvertierung ist vollständig implementiert; 
    sie verwendet keine BASIC-Routinen.

    Nach einem erfolglosen Zug wird das aktuelle Feld erneut ausgegeben. Erst wenn alle acht 
    Nachbarpaare aufsteigend sortiert sind, erscheint YOU WON IN ... MOVES.

    Nach dem Sieg kehrt das Programm zum Woz-Monitor zurück.

Einschränkung: Die Spielregeln, die Sortierprüfung und der Ablauf entsprechen dem BASIC-Listing. 
Die Zufallsfolge ist bewusst nicht bitgenau identisch mit RND(20) des Apple-I-Integer-BASIC. 
Außerdem ist der Quelltext hier nicht gegen einen realen Apple-I-Emulator assembliert und getestet 
worden; ich kann daher keine erfolgreiche Build- oder Laufzeitprüfung behaupten.

; ============================================================
; REVERSE - Apple I BASIC nach 6502 / ca65
;
; Ziel:       Apple I / Emulator mit Woz-Monitor
; Start:      $0300
; CPU:        NMOS 6502
; Toolchain:  ca65 / ld65 (cc65)
;
; Monitor:
;   $FFEF = ECHO
;   $FF00 = Monitor-Einstieg/Reset
;
; Apple-I-Hardware:
;   $D010 = KBD
;   $D011 = KBDCR
;
; Assemblieren:
;   ca65 -o reverse.o reverse.s
;   ld65 -C apple1.cfg -o reverse.bin reverse.o
;
; Laden des Binärabbilds nach $0300 und Start mit:
;   0300R
; ============================================================

.setcpu "6502"

ECHO    = $FFEF
MONITOR = $FF00
KBD     = $D010
KBDCR   = $D011
CR      = $0D

; ------------------------------------------------------------
; Zero Page
; ------------------------------------------------------------

.segment "ZEROPAGE"

ptr:        .res 2
i:          .res 1
j:          .res 1
n:          .res 1
tmp:        .res 1
digit:      .res 1
inputval:   .res 1
input_bad:  .res 1
digits_seen:.res 1
moves:      .res 1
seed:       .res 1
hundreds:   .res 1
tens:       .res 1

; ------------------------------------------------------------
; BSS
; ------------------------------------------------------------

.segment "BSS"

board:      .res 9
backup:     .res 9

; ------------------------------------------------------------
; CODE
; ------------------------------------------------------------
.org $300 ;<- manually inserted P.Sieg - see explanation down in text
.segment "CODE"

.export start

.proc start
    cld
    ldx #$FF
    txs

    lda #$A7
    sta seed

    lda #0
    sta moves

    lda #<title
    ldx #>title
    jsr print_string
    jsr newline

; ============================================================
; BASIC 100-180: Zufallspermutation erstellen
; ============================================================

new_game:
    ldx #0
    lda #0

clear_board:
    sta board,x
    inx
    cpx #9
    bne clear_board

    lda #1
    sta i

place_number:
    jsr random_0_19
    clc
    adc #1
    sta j

    ; Zufallszahl groesser als 9?
    cmp #10
    bcs place_number

    ; Index von 1..9 auf 0..8 umstellen
    sec
    sbc #1
    tax

    ; Position bereits belegt?
    lda board,x
    bne place_number

    ; Zahl eintragen
    lda i
    sta board,x

    inc i
    lda i
    cmp #10
    bne place_number

; ============================================================
; BASIC 190: Spielfeld ausgeben
; ============================================================

show_board:
    jsr print_board

; ============================================================
; BASIC 200: Umkehrlaenge abfragen
; ============================================================

ask_number:
    lda #<question
    ldx #>question
    jsr print_string

    jsr read_number
    sta n

    lda input_bad
    bne ask_number

; BASIC 210-220: N darf hoechstens 9 sein
    lda n
    cmp #10
    bcc check_minimum

    lda #<too_many
    ldx #>too_many
    jsr print_string
    jmp ask_number

; BASIC 230-241: N muss groesser als 1 sein
check_minimum:
    lda n
    cmp #2
    bcs reverse_prefix

    lda #<too_small
    ldx #>too_small
    jsr print_string
    jmp ask_number

; ============================================================
; BASIC 250: Zugzaehler erhoehen
; ============================================================

reverse_prefix:
    inc moves

; BASIC 260: B(I) = A(I), I=1..9
    ldx #0

copy_board:
    lda board,x
    sta backup,x
    inx
    cpx #9
    bne copy_board

; ============================================================
; BASIC 270:
; FOR I=1 TO N
;   J=N+1-I
;   A(J)=B(I)
; NEXT I
;
; Nullbasierte Indizes:
; board[N-1-i] = backup[i]
; ============================================================

    lda #0
    sta i

reverse_loop:
    lda n
    sec
    sbc #1
    sec
    sbc i
    tax

    ldy i
    lda backup,y
    sta board,x

    inc i
    lda i
    cmp n
    bcc reverse_loop

; ============================================================
; BASIC 280-310: Sortierpruefung
;
; Wenn board[i] > board[i+1], ist das Feld unsortiert.
; In diesem Fall zur Ausgabe zurueckkehren, ohne neu zu mischen.
; ============================================================

    ldx #0

check_order:
    lda board,x
    cmp board+1,x

    ; Links kleiner als rechts
    bcc next_pair

    ; Gleich
    beq next_pair

    ; Links groesser: noch nicht gewonnen
    jmp show_board

next_pair:
    inx
    cpx #8
    bne check_order

; ============================================================
; BASIC 320-330: Sieg
; ============================================================

    lda #<won_text
    ldx #>won_text
    jsr print_string

    lda moves
    jsr print_decimal

    lda #<moves_text
    ldx #>moves_text
    jsr print_string

    ; Zum Woz-Monitor zurueckkehren
    jmp MONITOR

.endproc

; ============================================================
; Zufallszahl 0..19
;
; 8-Bit-LFSR als Ersatz fuer BASIC-RND.
; Nicht identisch mit der BASIC-Zufallsfolge.
; ============================================================

.proc random_0_19
    lda seed
    asl a
    bcc random_no_xor
    eor #$1D

random_no_xor:
    bne random_save
    lda #$A7

random_save:
    sta seed

    and #$1F
    cmp #20
    bcc random_done

    sec
    sbc #20

random_done:
    rts
.endproc

; ============================================================
; Nullterminierten String ausgeben
;
; Eingang: A = Low-Byte, X = High-Byte
; ============================================================

.proc print_string
    sta ptr
    stx ptr+1

    ldy #0

print_string_loop:
    lda (ptr),y
    beq print_string_done

    jsr ECHO
    iny
    bne print_string_loop

print_string_done:
    rts
.endproc

; ============================================================
; Wagenruecklauf
; ============================================================

.proc newline
    lda #CR
    jsr ECHO
    rts
.endproc

; ============================================================
; Spielfeld ausgeben
; ============================================================

.proc print_board
    lda #<numbers
    ldx #>numbers
    jsr print_string
    jsr newline

    lda #<separator
    ldx #>separator
    jsr print_string
    jsr newline

    lda #'('
    jsr ECHO

    ldx #0

print_board_loop:
    lda board,x
    jsr print_decimal

    inx
    cpx #9
    beq print_board_end

    lda #' '
    jsr ECHO
    jmp print_board_loop

print_board_end:
    lda #')'
    jsr ECHO
    jsr newline

    lda #<separator2
    ldx #>separator2
    jsr print_string
    jsr newline
    jsr newline
    rts
.endproc

; ============================================================
; Dezimalzahl 0..255 ausgeben
; Ohne fuehrende Nullen.
; ============================================================

.proc print_decimal
    sta tmp

    lda #0
    sta hundreds
    sta tens

decimal_hundreds_loop:
    lda tmp
    cmp #100
    bcc decimal_tens_start

    sec
    sbc #100
    sta tmp

    inc hundreds
    jmp decimal_hundreds_loop

decimal_tens_start:
    lda #0
    sta tens

decimal_tens_loop:
    lda tmp
    cmp #10
    bcc decimal_output

    sec
    sbc #10
    sta tmp

    inc tens
    jmp decimal_tens_loop

decimal_output:
    ; Hunderter ausgeben, falls vorhanden
    lda hundreds
    beq decimal_check_tens

    clc
    adc #'0'
    jsr ECHO

    ; Nach Hundertern immer die Zehnerstelle ausgeben
    lda tens
    clc
    adc #'0'
    jsr ECHO
    jmp decimal_ones

decimal_check_tens:
    lda tens
    beq decimal_ones

    clc
    adc #'0'
    jsr ECHO

decimal_ones:
    lda tmp
    clc
    adc #'0'
    jsr ECHO
    rts
.endproc

; ============================================================
; Tastatureingabe einer Dezimalzahl
;
; Wartet auf RETURN.
; Ignoriert Leerzeichen.
; Nichtnumerische Zeichen machen die Eingabe ungueltig.
; Werte ueber 255 werden auf 255 begrenzt.
;
; Rueckgabe:
;   A         = Eingabewert
;   input_bad = 0 bei gueltiger Eingabe, sonst 1
; ============================================================

.proc read_number
    lda #0
    sta inputval
    sta input_bad
    sta digits_seen

read_number_char:
    ; Auf Tastendruck warten: Bit 7 von KBDCR
    lda KBDCR
    bpl read_number_char

    ; Apple-I-Tastatur lesen, hohes Bit entfernen
    lda KBD
    and #$7F

    ; RETURN?
    cmp #CR
    beq read_number_done

    ; Zeichen echoen
    jsr ECHO

    ; Leerzeichen ignorieren
    cmp #' '
    beq read_number_char

    ; Nur ASCII-Ziffern akzeptieren
    cmp #'0'
    bcc read_number_bad_char

    cmp #'9'+1
    bcs read_number_bad_char

    sec
    sbc #'0'
    sta digit
    inc digits_seen

    ; Pruefen, ob inputval*10+digit > 255 wird.
    ;
    ; inputval >= 26 -> immer groesser als 255
    ; inputval == 25 und digit >= 6 -> groesser als 255
    lda inputval
    cmp #26
    bcs read_number_saturate

    cmp #25
    bne read_number_multiply

    lda digit
    cmp #6
    bcs read_number_saturate

read_number_multiply:
    ; inputval * 10 = inputval * 8 + inputval * 2
    lda inputval
    asl a
    sta tmp                 ; 2*x

    asl a                   ; 4*x
    asl a                   ; 8*x

    clc
    adc tmp                 ; 10*x
    clc
    adc digit
    sta inputval

    jmp read_number_char

read_number_saturate:
    lda #255
    sta inputval
    jmp read_number_char

read_number_bad_char:
    lda #1
    sta input_bad
    jmp read_number_char

read_number_done:
    jsr newline

    ; Mindestens eine Ziffer erforderlich
    lda digits_seen
    bne read_number_return

    lda #1
    sta input_bad

read_number_return:
    lda inputval
    rts
.endproc

; ============================================================
; Texte
; ============================================================

.segment "RODATA"

title:
    .byte "R E V E R S E",0

numbers:
    .byte " 1 2 3 4 5 6 7 8 9",0

separator:
    .byte " -----------------",0

separator2:
    .byte "------------------",0

question:
    .byte "HOW MANY SHALL I REVERSE? ",0

too_many:
    .byte "OOPS! TOO MANY- I CAN REVERSE AT MOST 9",CR,0

too_small:
    .byte "OOPS ! TOO SMALL; YOU MUST REVERSE SOMETHING!",CR,0

won_text:
    .byte "YOU WON IN ",0

moves_text:
    .byte " MOVES",CR,0

Für ein einfaches Apple-I-Programm ist .org $0300 mit einem passenden Binärformat oft der 
unkompliziertere Weg. Achte nur darauf, dass dein Assembler tatsächlich ein rohes Binärabbild 
ab dieser Adresse erzeugt und du es im Monitor an $0300 lädst.
